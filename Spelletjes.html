    <!-- type="text" om te voorkomen dat het mobiele numerieke toetsenbord ongewenst verschijnt, 

         aangezien we nu ons eigen toetsenbord hebben. 'readonly' om ongewenste invoer te blokkeren. -->

<script>

    let multiplier;

    let score = 0;

    const tableNumber = 9;


    const questionDisplay = document.getElementById('questionDisplay');

    const answerInput = document.getElementById('answerInput');

    const currentScore = document.getElementById('currentScore');

    const feedbackBox = document.getElementById('feedbackBox');


    // Functie om een cijfer toe te voegen aan het invoerveld (via knoppen)

    function appendDigit(digit) {

        // Maximaal 3 cijfers is ruim voldoende voor 9 x 10 = 90

        if (answerInput.value.length < 3) { 

            answerInput.value += digit;

            // Verberg feedback bij nieuwe invoer

            hideFeedback();

        }

    }


    // Functie om het laatste cijfer te verwijderen (Backspace)

    function deleteDigit() {

        answerInput.value = answerInput.value.slice(0, -1);

        hideFeedback();

    }

    

    // Functie om de invoer volledig te wissen (Clear)

    function clearInput() {

        answerInput.value = '';

        hideFeedback();

    }


    // Functie om een nieuwe som te genereren

    function generateNewQuestion() {

        // Genereer een willekeurig getal van 1 t/m 10

        multiplier = Math.floor(Math.random() * 10) + 1;

        questionDisplay.textContent = `${tableNumber} x ${multiplier}`;

        answerInput.value = ''; 

        // Focus terugzetten voor visuele feedback, ondanks dat het veld readonly is

        answerInput.focus(); 

        hideFeedback();

    }


    // Functie om het antwoord te controleren

    function checkAnswer() {

        // Gebruik parseFloat of Integer voor numerieke conversie

        const userAnswer = parseInt(answerInput.value);

        const correctAnswer = tableNumber * multiplier;


        if (isNaN(userAnswer) || answerInput.value.trim() === '') {

            showFeedback("Voer alsjeblieft een nummer in.", 'incorrect');

            return;

        }


        if (userAnswer === correctAnswer) {

            score++;

            showFeedback("Juist! Goed gedaan!", 'correct');

            updateScore();

            // Geef een korte pauze voordat de nieuwe som komt

            setTimeout(generateNewQuestion, 1000);

        } else {

            showFeedback(`Fout. Het juiste antwoord is ${correctAnswer}. Probeer het nog een keer!`, 'incorrect');

            // Reset de invoer zodat de gebruiker opnieuw kan proberen

            answerInput.value = '';

            answerInput.focus();

        }

    }


    // Functie om de score bij te werken

    function updateScore() {

        currentScore.textContent = score;

    }


    // Functie voor feedbackweergave

    function showFeedback(message, type) {

        // Eerst alle feedbackklassen verwijderen

        feedbackBox.className = 'feedback-box';


        // Nieuwe klasse toevoegen op basis van het type

        if (type === 'correct') {

            feedbackBox.classList.add('feedback-correct');

        } else {

            feedbackBox.classList.add('feedback-incorrect');

        }


        feedbackBox.textContent = message;

        feedbackBox.style.opacity = 1; // Maak zichtbaar

    }


    // Functie om feedback te verbergen

    function hideFeedback() {

        feedbackBox.classList.remove('feedback-correct', 'feedback-incorrect');

        feedbackBox.textContent = '';

        feedbackBox.style.opacity = 0; 

    }


    // Behoud de ondersteuning voor de 'Enter' toets voor als er toch een fysiek toetsenbord wordt gebruikt.

    function handleKeydown(event) {

        if (event.key === 'Enter') { 

            checkAnswer();

            event.preventDefault(); 

        }

    }


    // Start het spel wanneer de pagina geladen is

    window.onload = function() {

        generateNewQuestion();

        updateScore();

    };

</script>
